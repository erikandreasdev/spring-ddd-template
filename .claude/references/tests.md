# Test references

An example `orders` context. Copy the shape (names, visibility, validation, `final`, structure), not the business: the domain is fictional.
The `//` comments explain the pattern for you; they are not copied into project code, which only carries comments about the why.
The code blocks are not compiled or run. The source of truth is the rules; if anything conflicts, the rules win.

Rules: `.claude/rules/testing.md`.

## Object Mother

`src/test/java/com/CHANGEME/orders/domain/OrderMother.java`

```java
package com.CHANGEME.orders.domain;

import com.CHANGEME.shared.domain.Money;
import java.math.BigDecimal;
import java.time.Instant;
import java.util.Currency;
import org.instancio.Instancio;

// Object Mother: one factory per scenario. Aggregates are always built with the business
// factories and methods; values the tests do not check (IDs, amounts, quantities, dates) are
// generated by Instancio within valid ranges.
// The variants that receive an OrderId are for integration tests, which request the ID from
// orders.nextId() so that it matches the sequence.
public final class OrderMother {

  public static final Currency EUR = Currency.getInstance("EUR");

  private OrderMother() {}

  public static OrderId id() {
    return new OrderId(positiveLong());
  }

  public static CustomerId customerId() {
    return new CustomerId(positiveLong());
  }

  // Valid random amount: between 0.01 and 1000.00, always in the same currency.
  public static Money price() {
    return new Money(BigDecimal.valueOf(Instancio.gen().ints().range(1, 100_000).get(), 2), EUR);
  }

  public static Order draft() {
    return draft(id(), customerId());
  }

  public static Order draft(final OrderId id, final CustomerId customerId) {
    return Order.place(id, customerId, EUR, Instancio.create(Instant.class));
  }

  public static Order draftWithOneLine() {
    return draftWithOneLine(id(), customerId());
  }

  public static Order draftWithOneLine(final OrderId id) {
    return draftWithOneLine(id, customerId());
  }

  public static Order draftWithOneLine(final OrderId id, final CustomerId customerId) {
    final Order order = draft(id, customerId);
    order.addLine(new ProductId(positiveLong()), price(), Instancio.gen().ints().range(1, 20).get());
    return order;
  }

  public static Order confirmed() {
    return confirmed(id());
  }

  public static Order confirmed(final OrderId id) {
    final Order order = draftWithOneLine(id);
    order.confirm();
    return order;
  }

  private static long positiveLong() {
    return Instancio.gen().longs().min(1L).get();
  }
}
```

    // then

    // given

import static com.github.tomakehurst.wiremock.client.WireMock.aResponse;

import static org.assertj.core.api.Assertions.assertThat;
import static org.assertj.core.api.Assertions.assertThatThrownBy;

import com.CHANGEME.IntegrationTest;
import com.CHANGEME.orders.domain.Order;
import com.CHANGEME.orders.domain.OrderMother;
import com.CHANGEME.orders.domain.OrderRepository;
import com.CHANGEME.orders.domain.OrderStatus;
import com.github.tomakehurst.wiremock.WireMockServer;
import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;

// Use case with an external service: the real adapter (HttpPaymentGateway) against WireMock.
// Never @MockitoBean or fakes on the port. The second test proves the rollback: the declined
// charge leaves the order as it was.
@IntegrationTest
class PayOrderServiceTest {

  @Autowired private PayOrderService payOrder;
  @Autowired private OrderRepository orders;
  @Autowired private WireMockServer paymentProvider;

  // Technical setup: @BeforeEach never creates business data.
  @BeforeEach
  void resetPaymentProvider() {
    paymentProvider.resetAll();
  }

  @Test
  void givenConfirmedOrder_whenPay_thenOrderIsPaidAndCharged() {
    // given
    final Order order = OrderMother.confirmed(orders.nextId());
    orders.save(order);
    paymentProvider.stubFor(post(urlEqualTo("/charges")).willReturn(aResponse().withStatus(201)));

    // when
    payOrder.pay(new PayOrderCommand(order.id()));

    // then
    assertThat(orders.findById(order.id()).orElseThrow().status()).isEqualTo(OrderStatus.PAID);
    paymentProvider.verify(
        postRequestedFor(urlEqualTo("/charges"))
            .withRequestBody(matchingJsonPath("$.reference", equalTo("order-" + order.id().value())))
            .withRequestBody(
                matchingJsonPath("$.amount", equalTo(order.total().amount().toPlainString()))));
  }

  @Test
  void givenDeclinedPayment_whenPay_thenOrderStaysConfirmed() {
    // given
    final Order order = OrderMother.confirmed(orders.nextId());
    orders.save(order);
    paymentProvider.stubFor(post(urlEqualTo("/charges")).willReturn(aResponse().withStatus(402)));

    // when / then
    assertThatThrownBy(() -> payOrder.pay(new PayOrderCommand(order.id())))
        .isInstanceOf(PaymentDeclinedException.class);
    assertThat(orders.findById(order.id()).orElseThrow().status())
        .isEqualTo(OrderStatus.CONFIRMED);
  }
}

import static com.CHANGEME.orders.domain.OrderMother.EUR;

// Read port test: one per filter, ordering or pagination. Each test uses a new customer, so
// data from other tests does not interfere even though the database is shared.
@IntegrationTest
class ListCustomerOrdersServiceTest {

  @Autowired private ListCustomerOrdersService listCustomerOrders;
  @Autowired private OrderRepository orders;

  @Test
  void givenOrdersOfSeveralCustomers_whenList_thenOnlyThatCustomerOrdersAreCounted() {
    // given
    final CustomerId customer = OrderMother.customerId();
    orders.save(OrderMother.draftWithOneLine(orders.nextId(), customer));
    orders.save(OrderMother.draftWithOneLine(orders.nextId(), customer));
    orders.save(OrderMother.draftWithOneLine(orders.nextId(), OrderMother.customerId()));

    // when
    final PageResult<OrderSummary> result =
        listCustomerOrders.list(new ListCustomerOrdersQuery(customer, 0, 20));

    // then
    assertThat(result.totalItems()).isEqualTo(2);
    assertThat(result.items()).extracting(OrderSummary::customerId).containsOnly(customer);
  }

  @Test
  void givenThreeOrders_whenListSecondPageOfTwo_thenOldestOrderIsReturned() {
    // given
    final CustomerId customer = OrderMother.customerId();
    final OrderId oldest = placeAt(customer, "2026-01-01T10:00:00Z");
    placeAt(customer, "2026-02-01T10:00:00Z");
    placeAt(customer, "2026-03-01T10:00:00Z");

    // when
    final PageResult<OrderSummary> result =
        listCustomerOrders.list(new ListCustomerOrdersQuery(customer, 1, 2));

    // then
    assertThat(result.items()).extracting(OrderSummary::id).containsExactly(oldest);
    assertThat(result.totalItems()).isEqualTo(3);
  }

  // Here the date is a literal because the ordering depends on it.
  private OrderId placeAt(final CustomerId customer, final String placedAt) {
    final Order order = Order.place(orders.nextId(), customer, EUR, Instant.parse(placedAt));
    orders.save(order);
    return order.id();
  }
}

  @Test
  void givenTwoCopiesOfTheSameVersion_whenBothAreSaved_thenSecondSaveFails() {

    final Order second = orders.findById(order.id()).orElseThrow();

// Single WireMock server for all integration tests, like TestcontainersConfig.

//   1. test dependency org.wiremock:wiremock-standalone (3.13.2 or the latest 3.x)
//   2. @Import({TestcontainersConfig.class, WireMockConfig.class}) in IntegrationTest
